import { render, screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import type { UserEvent } from '@testing-library/user-event'
import { beforeAll, describe, expect, it, vi } from 'vitest'

import { DisplayPreferencesStudy } from '../display-preferences-study'

// Radix Select reads pointer-capture and scroll APIs that jsdom does not implement.
beforeAll(() => {
  Element.prototype.scrollIntoView = vi.fn()
  Element.prototype.hasPointerCapture = vi.fn(() => false)
  Element.prototype.setPointerCapture = vi.fn()
  Element.prototype.releasePointerCapture = vi.fn()
})

const NUMBER_FORMAT_TRIGGER = 'display-preferences-number-format'
const TIME_ZONE_TRIGGER = 'display-preferences-time-zone'

const trigger = (id: string) => {
  const control = document.getElementById(id)

  expect(control).toBeInTheDocument()

  return control as HTMLElement
}

const chooseOption = async (user: UserEvent, id: string, option: string) => {
  await user.click(trigger(id))
  await user.click(await screen.findByRole('option', { name: option }))
}

describe('display preferences study', () => {
  it('starts on the first option of each labelled select with no saved feedback', () => {
    render(<DisplayPreferencesStudy />)

    expect(screen.getByText('Number format').closest('label')).toHaveAttribute(
      'for',
      NUMBER_FORMAT_TRIGGER
    )
    expect(screen.getByText('Time zone').closest('label')).toHaveAttribute(
      'for',
      TIME_ZONE_TRIGGER
    )
    expect(trigger(NUMBER_FORMAT_TRIGGER)).toHaveTextContent('1,234.56')
    expect(trigger(TIME_ZONE_TRIGGER)).toHaveTextContent('Local time')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
  })

  it('keeps saved values for the session and drops the feedback on the next change', async () => {
    const user = userEvent.setup()

    render(<DisplayPreferencesStudy />)

    const study = screen.getByTestId('display-preferences-study')

    await chooseOption(user, NUMBER_FORMAT_TRIGGER, '1.234,56')
    await chooseOption(user, TIME_ZONE_TRIGGER, 'UTC')

    expect(study).not.toHaveAttribute('data-saved-number-format')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()

    await user.click(screen.getByRole('button', { name: 'Save preferences' }))

    expect(study).toHaveAttribute('data-saved-number-format', 'comma-decimal')
    expect(study).toHaveAttribute('data-saved-time-zone', 'utc')
    expect(screen.getByRole('status')).toHaveTextContent('Preferences saved')

    await chooseOption(user, TIME_ZONE_TRIGGER, 'Local time')

    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
    expect(trigger(NUMBER_FORMAT_TRIGGER)).toHaveTextContent('1.234,56')
  })

  it('returns both selects to their first option on reset', async () => {
    const user = userEvent.setup()

    render(<DisplayPreferencesStudy />)

    await chooseOption(user, NUMBER_FORMAT_TRIGGER, '1.234,56')
    await chooseOption(user, TIME_ZONE_TRIGGER, 'UTC')
    await user.click(screen.getByRole('button', { name: 'Save preferences' }))
    await user.click(screen.getByRole('button', { name: 'Reset' }))

    expect(trigger(NUMBER_FORMAT_TRIGGER)).toHaveTextContent('1,234.56')
    expect(trigger(TIME_ZONE_TRIGGER)).toHaveTextContent('Local time')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
  })
})
