import userEvent from '@testing-library/user-event'
import { render, screen } from '@testing-library/react'
import { beforeAll, describe, expect, it, vi } from 'vitest'

import { DisplayPreferencesStudy } from '../display-preferences-study'

// Radix popups need pointer-capture and scroll APIs jsdom does not implement.
beforeAll(() => {
  window.HTMLElement.prototype.hasPointerCapture = vi.fn(() => false)
  window.HTMLElement.prototype.setPointerCapture = vi.fn()
  window.HTMLElement.prototype.releasePointerCapture = vi.fn()
  window.HTMLElement.prototype.scrollIntoView = vi.fn()
})

const choose = async (
  user: ReturnType<typeof userEvent.setup>,
  label: string,
  option: string
) => {
  await user.click(screen.getByLabelText(label))
  await user.click(await screen.findByRole('option', { name: option }))
}

const panel = () => screen.getByTestId('display-preferences-study')

describe('display preferences study', () => {
  it('starts on the first option of each select with no saved feedback', () => {
    render(<DisplayPreferencesStudy />)

    expect(screen.getByLabelText('Number format')).toHaveTextContent('1,234.56')
    expect(screen.getByLabelText('Time zone')).toHaveTextContent('Local time')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
    expect(panel()).not.toHaveAttribute('data-saved-number-format')
  })

  it('stores the selected values for the session and confirms the save', async () => {
    const user = userEvent.setup()

    render(<DisplayPreferencesStudy />)

    await choose(user, 'Number format', '1.234,56')
    await choose(user, 'Time zone', 'UTC')
    await user.click(screen.getByRole('button', { name: 'Save preferences' }))

    expect(screen.getByText('Preferences saved')).toBeVisible()
    expect(panel()).toHaveAttribute('data-saved-number-format', '1.234,56')
    expect(panel()).toHaveAttribute('data-saved-time-zone', 'UTC')
  })

  it('clears the saved feedback when either select changes', async () => {
    const user = userEvent.setup()

    render(<DisplayPreferencesStudy />)

    await user.click(screen.getByRole('button', { name: 'Save preferences' }))
    expect(screen.getByText('Preferences saved')).toBeVisible()

    await choose(user, 'Time zone', 'UTC')

    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
    expect(screen.getByLabelText('Time zone')).toHaveTextContent('UTC')

    await user.click(screen.getByRole('button', { name: 'Save preferences' }))
    expect(screen.getByText('Preferences saved')).toBeVisible()

    await choose(user, 'Number format', '1.234,56')

    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
  })

  it('restores both initial options and clears feedback on reset', async () => {
    const user = userEvent.setup()

    render(<DisplayPreferencesStudy />)

    await choose(user, 'Number format', '1.234,56')
    await choose(user, 'Time zone', 'UTC')
    await user.click(screen.getByRole('button', { name: 'Save preferences' }))
    await user.click(screen.getByRole('button', { name: 'Reset' }))

    expect(screen.getByLabelText('Number format')).toHaveTextContent('1,234.56')
    expect(screen.getByLabelText('Time zone')).toHaveTextContent('Local time')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
  })

  it('keeps the saved values out of persistent storage', async () => {
    const user = userEvent.setup()
    const setItem = vi.spyOn(Storage.prototype, 'setItem')

    const { unmount } = render(<DisplayPreferencesStudy />)

    await choose(user, 'Number format', '1.234,56')
    await user.click(screen.getByRole('button', { name: 'Save preferences' }))

    expect(setItem).not.toHaveBeenCalled()

    unmount()
    render(<DisplayPreferencesStudy />)

    expect(screen.getByLabelText('Number format')).toHaveTextContent('1,234.56')
    expect(screen.queryByText('Preferences saved')).not.toBeInTheDocument()
    expect(panel()).not.toHaveAttribute('data-saved-number-format')

    setItem.mockRestore()
  })
})
