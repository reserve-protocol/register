import { useState } from 'react'

import { Button } from '@/components/button'
import { ActionGroup } from '@/components/design-system-v1/action-group'
import { Field, FieldLabel } from '@/components/design-system-v1/field'
import {
  InlineMessage,
  InlineMessageTitle,
} from '@/components/design-system-v1/inline-message'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/design-system-v1/select'
import { v1Typography } from '@/components/design-system-v1/typography'
import { v1LayoutRecipes } from '@/components/ui/v1-layout-recipes'
import { v1SemanticRecipes as roles } from '@/components/ui/v1-semantic-recipes'
import { cn } from '@/lib/utils'

const NUMBER_FORMAT_OPTIONS = [
  { value: 'dot-decimal', label: '1,234.56' },
  { value: 'comma-decimal', label: '1.234,56' },
] as const

const TIME_ZONE_OPTIONS = [
  { value: 'local', label: 'Local time' },
  { value: 'utc', label: 'UTC' },
] as const

type NumberFormat = (typeof NUMBER_FORMAT_OPTIONS)[number]['value']
type TimeZone = (typeof TIME_ZONE_OPTIONS)[number]['value']

interface DisplayPreferences {
  numberFormat: NumberFormat
  timeZone: TimeZone
}

const INITIAL_PREFERENCES: DisplayPreferences = {
  numberFormat: NUMBER_FORMAT_OPTIONS[0].value,
  timeZone: TIME_ZONE_OPTIONS[0].value,
}

export const DisplayPreferencesStudy = () => {
  const [numberFormat, setNumberFormat] = useState<NumberFormat>(
    INITIAL_PREFERENCES.numberFormat
  )
  const [timeZone, setTimeZone] = useState<TimeZone>(
    INITIAL_PREFERENCES.timeZone
  )
  const [sessionPreferences, setSessionPreferences] =
    useState<DisplayPreferences | null>(null)
  const [savedFeedbackVisible, setSavedFeedbackVisible] = useState(false)

  const selectNumberFormat = (value: string) => {
    setNumberFormat(value as NumberFormat)
    setSavedFeedbackVisible(false)
  }

  const selectTimeZone = (value: string) => {
    setTimeZone(value as TimeZone)
    setSavedFeedbackVisible(false)
  }

  const savePreferences = () => {
    setSessionPreferences({ numberFormat, timeZone })
    setSavedFeedbackVisible(true)
  }

  const resetPreferences = () => {
    setNumberFormat(INITIAL_PREFERENCES.numberFormat)
    setTimeZone(INITIAL_PREFERENCES.timeZone)
    setSavedFeedbackVisible(false)
  }

  return (
    <section
      data-testid="display-preferences-study"
      data-saved-number-format={sessionPreferences?.numberFormat}
      data-saved-time-zone={sessionPreferences?.timeZone}
      aria-labelledby="display-preferences-study-title"
      className={cn(
        'mx-auto w-full max-w-xl border border-border bg-card p-4 sm:p-6',
        v1LayoutRecipes.stack.completeGroups
      )}
    >
      <header className={v1LayoutRecipes.stack.tightText}>
        <h2
          id="display-preferences-study-title"
          className="text-xl font-medium leading-7"
        >
          Display preferences
        </h2>
        <p className={cn(v1Typography.supporting, roles.text.supporting)}>
          Choose how amounts and times appear.
        </p>
      </header>

      <div
        className={cn(
          'grid sm:grid-cols-2',
          v1LayoutRecipes.cluster.internalRegions
        )}
      >
        <Field>
          <FieldLabel htmlFor="display-preferences-number-format">
            Number format
          </FieldLabel>
          <Select value={numberFormat} onValueChange={selectNumberFormat}>
            <SelectTrigger id="display-preferences-number-format">
              <SelectValue />
            </SelectTrigger>
            <SelectContent>
              {NUMBER_FORMAT_OPTIONS.map((option) => (
                <SelectItem key={option.value} value={option.value}>
                  {option.label}
                </SelectItem>
              ))}
            </SelectContent>
          </Select>
        </Field>

        <Field>
          <FieldLabel htmlFor="display-preferences-time-zone">
            Time zone
          </FieldLabel>
          <Select value={timeZone} onValueChange={selectTimeZone}>
            <SelectTrigger id="display-preferences-time-zone">
              <SelectValue />
            </SelectTrigger>
            <SelectContent>
              {TIME_ZONE_OPTIONS.map((option) => (
                <SelectItem key={option.value} value={option.value}>
                  {option.label}
                </SelectItem>
              ))}
            </SelectContent>
          </Select>
        </Field>
      </div>

      <div
        className={cn(
          'flex flex-col sm:flex-row sm:items-center sm:justify-between',
          v1LayoutRecipes.cluster.internalRegions
        )}
      >
        <ActionGroup
          direction="vertical"
          className="sm:w-auto sm:flex-row sm:[&>*]:w-auto"
        >
          <Button onClick={savePreferences}>Save preferences</Button>
          <Button tone="secondary" onClick={resetPreferences}>
            Reset
          </Button>
        </ActionGroup>
        <div role="status" aria-live="polite" className="min-w-0">
          {savedFeedbackVisible && (
            <InlineMessage
              data-testid="display-preferences-saved"
              presentation="summary"
              tone="success"
            >
              <InlineMessageTitle>Preferences saved</InlineMessageTitle>
            </InlineMessage>
          )}
        </div>
      </div>
    </section>
  )
}

export default DisplayPreferencesStudy
