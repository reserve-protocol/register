import { fireEvent, render, screen } from '@testing-library/react'
import { describe, expect, it } from 'vitest'

import { DisplayPreferencesStudy } from '../display-preferences-study'

type PreferenceField = 'number-format' | 'time-zone'

const preferenceTrigger = (field: PreferenceField) => {
  const element = document.getElementById(`display-preferences-${field}`)
  expect(element).toBeInstanceOf(HTMLButtonElement)
  return element as HTMLButtonElement
}

// Keyboard path only: Radix opens on ArrowDown and selects on Enter, which
// avoids the pointer-capture APIs jsdom does not implement.
const chooseOption = async (field: PreferenceField, optionLabel: string) => {
  fireEvent.keyDown(preferenceTrigger(field), { key: 'ArrowDown' })
  fireEvent.keyDown(await screen.findByRole('option', { name: optionLabel }), {
    key: 'Enter',
  })
}

const savedFeedback = () => screen.queryByText('Preferences saved')

describe('display preferences study', () => {
  it('starts on the first option of each select with no saved feedback', () => {
    render(<DisplayPreferencesStudy />)

    expect(
      screen.getByRole('heading', { name: 'Display preferences' })
    ).toBeVisible()
    expect(
      screen.getByText('Choose how amounts and times appear.')
    ).toBeVisible()
    expect(screen.getByText('Number format')).toHaveAttribute(
      'for',
      'display-preferences-number-format'
    )
    expect(screen.getByText('Time zone')).toHaveAttribute(
      'for',
      'display-preferences-time-zone'
    )
    expect(preferenceTrigger('number-format')).toHaveTextContent('1,234.56')
    expect(preferenceTrigger('time-zone')).toHaveTextContent('Local time')
    expect(savedFeedback()).not.toBeInTheDocument()
    expect(
      screen.getByRole('button', { name: 'Save preferences' })
    ).toBeVisible()
    expect(screen.getByRole('button', { name: 'Reset' })).toBeVisible()
  })

  it('saves the selected values for the mounted session and confirms', async () => {
    render(<DisplayPreferencesStudy />)

    await chooseOption('time-zone', 'UTC')
    fireEvent.click(screen.getByRole('button', { name: 'Save preferences' }))

    const study = screen.getByTestId('display-preferences-study')
    expect(preferenceTrigger('time-zone')).toHaveTextContent('UTC')
    expect(study).toHaveAttribute('data-stored-preferences', 'comma-group:utc')
    expect(savedFeedback()).toBeVisible()
    expect(screen.getByTestId('display-preferences-status')).toContainElement(
      savedFeedback()
    )
    // The specimen keeps the values in component state only.
    expect(localStorage.length).toBe(0)
    expect(sessionStorage.length).toBe(0)
  })

  it('clears the saved feedback when either select changes', async () => {
    render(<DisplayPreferencesStudy />)

    const save = screen.getByRole('button', { name: 'Save preferences' })

    fireEvent.click(save)
    expect(savedFeedback()).toBeVisible()
    await chooseOption('number-format', '1.234,56')
    expect(savedFeedback()).not.toBeInTheDocument()

    fireEvent.click(save)
    expect(savedFeedback()).toBeVisible()
    await chooseOption('time-zone', 'UTC')
    expect(savedFeedback()).not.toBeInTheDocument()
  })

  it('restores the initial options and clears the feedback on reset', async () => {
    render(<DisplayPreferencesStudy />)

    await chooseOption('number-format', '1.234,56')
    await chooseOption('time-zone', 'UTC')
    fireEvent.click(screen.getByRole('button', { name: 'Save preferences' }))
    expect(savedFeedback()).toBeVisible()

    fireEvent.click(screen.getByRole('button', { name: 'Reset' }))

    expect(preferenceTrigger('number-format')).toHaveTextContent('1,234.56')
    expect(preferenceTrigger('time-zone')).toHaveTextContent('Local time')
    expect(savedFeedback()).not.toBeInTheDocument()
  })
})
